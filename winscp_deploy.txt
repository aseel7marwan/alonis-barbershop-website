# WinSCP Script for Aloni's Barber Shop Deployment
# -------------------------------------------------

# Automatically answer all prompts
option batch on
# Disable overwriting confirmations
option confirm off

# Connect to the server
# Credentials and host are provided via environment variables loaded from .env:
#   DEPLOY_SFTP_USER, DEPLOY_SFTP_PASS, DEPLOY_SFTP_HOST, DEPLOY_SFTP_HOSTKEY
# Format with env vars: sftp://%DEPLOY_SFTP_USER%:%DEPLOY_SFTP_PASS%@%DEPLOY_SFTP_HOST%/
open sftp://%DEPLOY_SFTP_USER%:%DEPLOY_SFTP_PASS%@%DEPLOY_SFTP_HOST%/ -hostkey="%DEPLOY_SFTP_HOSTKEY%"

# Synchronize local files to remote server
# '|' starts the exclusion list. 
# We exclude the master plan as requested.
synchronize remote . ./ -filemask="|antigravity_master_plan.md; winscp_deploy.txt; deploy.bat; deploy.log; .git/; .agent/"

# Disconnect
close
# Exit WinSCP
exit
